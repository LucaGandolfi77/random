cmake_minimum_required(VERSION 3.16)
project(real-time-embedded-ai CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(rteai_tests src/tests_main.cpp src/runtime.cpp src/sha256.cpp)
target_include_directories(rteai_tests PRIVATE include models)
target_link_libraries(rteai_tests PRIVATE pthread)
add_executable(rteai_bench src/bench_main.cpp src/runtime.cpp src/sha256.cpp)
target_include_directories(rteai_bench PRIVATE include models)
target_link_libraries(rteai_bench PRIVATE pthread)
enable_testing()
add_test(NAME rteai_tests COMMAND rteai_tests)
